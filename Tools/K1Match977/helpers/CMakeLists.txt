cmake_minimum_required(VERSION 3.16)
project(K1BoundHelpers LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  message(FATAL_ERROR "Linux ARM64 cross target required")
endif()
set(SDK "${BHUMAN_ROOT}/Util/BoosterRoboticsSDK")
find_package(Threads REQUIRED)
add_executable(k1_mode_query mode_query.cpp)
add_executable(k1_damping_only damping_only.cpp)
add_executable(k1_dds_observer "${BHUMAN_ROOT}/Tools/K1Commissioning/dds_observer.cpp")
foreach(t k1_mode_query k1_damping_only k1_dds_observer)
  target_compile_features(${t} PRIVATE cxx_std_20)
  target_include_directories(${t} SYSTEM PRIVATE "${SDK}/include")
  target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(${t} PRIVATE "${SDK}/lib/LinuxArm64/libbooster_robotics_sdk.a"
    "${SDK}/lib/LinuxArm64/libfastrtps.so" "${SDK}/lib/LinuxArm64/libfastcdr.so"
    "${SDK}/lib/LinuxArm64/libfoonathan_memory-0.7.3.a" Threads::Threads)
  set_target_properties(${t} PROPERTIES SKIP_BUILD_RPATH TRUE)
endforeach()
